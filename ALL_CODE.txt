namespace RH {
  // ---------- storage helpers (wrapped in try/catch: storage can be blocked) ----------
  export function load<T>(key: string, fallback: T): T {
    try {
      const raw = localStorage.getItem("ridehub." + key);
      return raw ? (JSON.parse(raw) as T) : fallback;
    } catch {
      return fallback;
    }
  }

  export function save<T>(key: string, value: T): void {
    try {
      localStorage.setItem("ridehub." + key, JSON.stringify(value));
    } catch {
      /* ignore */
    }
  }

  // ---------- small utilities ----------
  export function esc(text: string): string {
    return text.replace(/[&<>"']/g, (c) =>
      ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c] as string)
    );
  }

  export function $(sel: string, root: ParentNode = document): HTMLElement {
    return root.querySelector(sel) as HTMLElement;
  }

  export function $$(sel: string, root: ParentNode = document): HTMLElement[] {
    return Array.from(root.querySelectorAll(sel)) as HTMLElement[];
  }

  export function isoDaysAgo(n: number): string {
    const d = new Date();
    d.setDate(d.getDate() - n);
    return d.toISOString().slice(0, 10);
  }

  export function inr(n: number): string {
    return "₹" + n.toLocaleString("en-IN");
  }

  export function fmtDuration(mins: number): string {
    const h = Math.floor(mins / 60);
    const m = mins % 60;
    return h ? `${h}h ${m.toString().padStart(2, "0")}m` : `${m}m`;
  }

  }

(function () {
  const { $, esc, fmtDuration, load, save, toast } = RH;

  // ---------- greeting ----------
  const hour = new Date().getHours();
  const greet = hour < 12 ? "Good morning" : hour < 18 ? "Good afternoon" : "Good evening";
  $("#greeting").textContent = `${greet}, rider!`;

  // ---------- headline stats ----------
  const totalKm = rides.reduce((s, r) => s + r.km, 0);
  const totalMins = rides.reduce((s, r) => s + r.mins, 0);
  const calories = Math.round(totalKm * 28); // rough estimate: ~28 kcal per km
  const avgSpeed = totalMins ? totalKm / (totalMins / 60) : 0;

  $("#stat-km").textContent = totalKm.toFixed(0);
  $("#stat-rides").textContent = String(rides.length);
  $("#stat-time").textContent = fmtDuration(totalMins);
  $("#stat-speed").textContent = avgSpeed.toFixed(1);
  $("#stat-cal").textContent = calories.toLocaleString("en-IN");

  // ---------- weekly bar chart (SVG, last 7 days) ----------
  function drawWeekChart(): void {
    const days: { label: string; km: number }[] = [];
    for (let i = 6; i >= 0; i--) {
      const iso = RH.isoDaysAgo(i);
      const d = new Date(iso + "T00:00:00");
      const km = rides.filter((r) => r.date === iso).reduce((s, r) => s + r.km, 0);
      days.push({ label: d.toLocaleDateString("en-US", { weekday: "short" }), km });
    }
    const max = Math.max(10, ...days.map((d) => d.km));
    const W = 560, H = 220, padB = 32, padT = 22, gap = 14;
    const bw = (W - gap * (days.length + 1)) / days.length;
    let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="Distance ridden in the last 7 days">`;
    [0.5, 1].forEach((f) => {
      const y = padT + (H - padT - padB) * (1 - f);
      svg += `<line class="grid" x1="0" x2="${W}" y1="${y}" y2="${y}"/>`;
    });
    days.forEach((d, i) => {
      const h = ((H - padT - padB) * d.km) / max;
      const x = gap + i * (bw + gap);
      const y = H - padB - h;
      svg += `<rect class="bar${d.km ? "" : " empty"}" x="${x}" y="${d.km ? y : H - padB - 3}" width="${bw}" height="${d.km ? h : 3}" rx="6"><title>${d.label}: ${d.km} km</title></rect>`;
      if (d.km) svg += `<text class="val" x="${x + bw / 2}" y="${y - 6}" text-anchor="middle">${d.km}</text>`;
      svg += `<text class="lbl" x="${x + bw / 2}" y="${H - 10}" text-anchor="middle">${d.label}</text>`;
    });
    svg += "</svg>";
    $("#week-chart").innerHTML = svg;
    const weekKm = days.reduce((s, d) => s + d.km, 0);
    $("#week-total").textContent = `${weekKm} km this week`;
  }
  drawWeekChart();

  // ---------- monthly goal ring ----------
  function drawGoal(): void {
    const goal = load<number>("goal", 300);
    const now = new Date();
    const monthKm = rides
      .filter((r) => {
        const d = new Date(r.date + "T00:00:00");
        return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear();
      })
      .reduce((s, r) => s + r.km, 0);
    const pct = Math.min(100, Math.round((monthKm / goal) * 100));
    const R = 54, C = 2 * Math.PI * R;
    $("#goal-ring").innerHTML =
      `<svg viewBox="0 0 140 140" role="img" aria-label="${pct}% of monthly goal">` +
      `<circle class="track" cx="70" cy="70" r="${R}"/>` +
      `<circle class="prog" cx="70" cy="70" r="${R}" stroke-dasharray="${C}" stroke-dashoffset="${C * (1 - pct / 100)}" transform="rotate(-90 70 70)"/>` +
      `<text x="70" y="68" text-anchor="middle" class="big">${pct}%</text>` +
      `<text x="70" y="88" text-anchor="middle" class="small">${monthKm} / ${goal} km</text></svg>`;
    ($("#goal-input") as HTMLInputElement).value = String(goal);
  }
  drawGoal();
  $("#goal-form").addEventListener("submit", (e) => {
    e.preventDefault();
    const v = Number(($("#goal-input") as HTMLInputElement).value);
    if (v >= 20 && v <= 5000) {
      save("goal", v);
      drawGoal();
      toast("Monthly goal updated");
    } else {
      toast("Enter a goal between 20 and 5000 km");
    }
  });

  // ---------- maintenance tracker ----------
  const parts = [
    { key: "chain", name: "Chain lube", every: 300 },
    { key: "brakes", name: "Brake pads check", every: 1000 },
    { key: "tyres", name: "Tyre replacement", every: 3000 },
  ];
  function drawMaintenance(): void {
    const base = load<Record<string, number>>("service", {});
    $("#maint-list").innerHTML = parts
      .map((p) => {

/* RideHub - Ride Log: add / delete rides, live summary, CSV export */
(function () {
  const { $, esc, fmtDuration, toast } = RH;
  RH.init("ridelog");

  let rides: Ride[] = RH.getRides();

  const form = $("#ride-form") as HTMLFormElement;
  const dateEl = $("#r-date") as HTMLInputElement;
  const routeEl = $("#r-route") as HTMLInputElement;
  const kmEl = $("#r-km") as HTMLInputElement;
  const minsEl = $("#r-mins") as HTMLInputElement;
  const bikeEl = $("#r-bike") as HTMLSelectElement;
  const preview = $("#speed-preview");

  dateEl.value = RH.isoDaysAgo(0);
  dateEl.max = RH.isoDaysAgo(0);

  function updatePreview(): void {
    const km = Number(kmEl.value);
    const mins = Number(minsEl.value);
    preview.textContent = km > 0 && mins > 0 ? `Average speed: ${(km / (mins / 60)).toFixed(1)} km/h` : "Average speed will show here";
  }
  kmEl.addEventListener("input", updatePreview);
  minsEl.addEventListener("input", updatePreview);

  function setError(id: string, msg: string): void {
    $("#err-" + id).textContent = msg;
  }

  function validate(): boolean {
    let ok = true;
    setError("route", "");
    setError("km", "");
    setError("mins", "");
    if (routeEl.value.trim().length < 3) {
      setError("route", "Enter a route name (min 3 characters)");
      ok = false;
    }
    const km = Number(kmEl.value);
    if (!(km > 0 && km <= 500)) {
      setError("km", "Distance must be between 0.1 and 500 km");
      ok = false;
    }
    const mins = Number(minsEl.value);
    if (!(mins >= 5 && mins <= 1500)) {
      setError("mins", "Duration must be 5 - 1500 minutes");
      ok = false;
    }
    return ok;
  }

  form.addEventListener("submit", (e) => {
    e.preventDefault();
    if (!validate()) return;
    const ride: Ride = {
      id: Date.now(),
      date: dateEl.value || RH.isoDaysAgo(0),
      route: routeEl.value.trim(),
      km: Math.round(Number(kmEl.value) * 10) / 10,
      mins: Math.round(Number(minsEl.value)),
      bike: bikeEl.value,
    };
    rides.push(ride);
    RH.saveRides(rides);
    form.reset();
    dateEl.value = RH.isoDaysAgo(0);
    updatePreview();
    render();
    toast("Ride saved!");
  });

  function render(): void {
    const sorted = [...rides].sort((a, b) => b.date.localeCompare(a.date) || b.id - a.id);
    const km = rides.reduce((s, r) => s + r.km, 0);
    const mins = rides.reduce((s, r) => s + r.mins, 0);
    $("#sum-rides").textContent = String(rides.length);
    $("#sum-km").textContent = km.toFixed(1);
    $("#sum-time").textContent = fmtDuration(mins);
    $("#sum-best").textContent = rides.length ? Math.max(...rides.map((r) => r.km)) + " km" : "-";

    $("#log-body").innerHTML = sorted.length
      ? sorted
          .map(
            (r) =>
              `<tr><td>${r.date}</td><td>${esc(r.route)}</td><td>${esc(r.bike)}</td><td>${r.km} km</td><td>${fmtDuration(r.mins)}</td>` +
              `<td>${(r.km / (r.mins / 60)).toFixed(1)}</td>` +
              `<td><button class="link danger" data-del="${r.id}" aria-label="Delete ride on ${r.date}">Delete</button></td></tr>`
          )
          .join("")
      : '<tr><td colspan="7" class="empty">No rides yet. Add your first ride above!</td></tr>';
  }

  $("#log-body").addEventListener("click", (e) => {
    const btn = (e.target as HTMLElement).closest("[data-del]") as HTMLElement | null;
    if (!btn) return;
    rides = rides.filter((r) => r.id !== Number(btn.dataset.del));
    RH.saveRides(rides);
    render();
    toast("Ride deleted");
  });

  $("#export").addEventListener("click", () => {
    const rows = [["Date", "Route", "Bike", "Distance_km", "Duration_min"]].concat(
      rides.map((r) => [r.date, '"' + r.route.replace(/"/g, '""') + '"', r.bike, String(r.km), String(r.mins)])
    );
    const blob = new Blob([rows.map((r) => r.join(",")).join("\n")], { type: "text/csv" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "ridehub-rides.csv";
    a.click();
    URL.revokeObjectURL(a.href);
    toast("CSV downloaded");
  });

  $("#reset").addEventListener("click", () => {
    if (confirm("Reset the log back to the demo rides?")) {
      try {
        localStorage.removeItem("ridehub.rides");
      } catch {
        /* ignore */
      }
      rides = RH.getRides();
      render();
      toast("Demo data restored");
    }
  });

  updatePreview();
  render();
})();

(function () {
  const { $, load, save, toast } = RH;
  RH.init("contact");

  const form = $("#join-form") as HTMLFormElement;
  const fields = {
    name: $("#c-name") as HTMLInputElement,
    email: $("#c-email") as HTMLInputElement,

  };

    el.setAttribute("aria-invalid", msg ? "true" : "false");

  function validate(): boolean {
    let ok = true;

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(fields.email.value.trim())) { setErr("email", "Enter a valid email address"); ok = false; } else setErr("email", "");
    if (fields.phone.value && !/^[0-9+\-\s]{10,14}$/.test(fields.phone.value.trim())) { setErr("phone", "Phone must be 10-14 digits"); ok = false; } else setErr("phone", "");
    if (!fields.level.value) { setErr("level", "Choose your riding level"); ok = false; } else setErr("level", "");
    if (fields.msg.value.trim().length < 10) { setErr("msg", "Tell us a little more (min 10 characters)"); ok = false; } else setErr("msg", "");
    return ok;
  }

  (Object.values(fields) as HTMLElement[]).forEach((el) => el.addEventListener("blur", () => { if (form.dataset.touched) validate(); }));

  const count = $("#msg-count");
  fields.msg.addEventListener("input", () => (count.textContent = `${fields.msg.value.length}/300`));

  form.addEventListener("submit", (e) => {
    e.preventDefault();
    form.dataset.touched = "1";
    if (!validate()) {
      toast("Please fix the highlighted fields");
      return;
    }
    const members = load<object[]>("members", []);
    members.push({ name: fields.name.value.trim(), email: fields.email.value.trim(), level: fields.level.value, at: new Date().toISOString() });
    save("members", members);
    form.hidden = true;
    const ok = $("#success");
    ok.hidden = false;
    $("#success-name").textContent = fields.name.value.trim().split(" ")[0];
    toast("Welcome to RideHub!");
  });

  // FAQ accordion
  document.querySelectorAll<HTMLButtonElement>(".faq button").forEach((btn) => {
    btn.addEventListener("click", () => {
      const open = btn.getAttribute("aria-expanded") === "true";
      btn.setAttribute("aria-expanded", String(!open));
      (btn.nextElementSibling as HTMLElement).hidden = open;
    });
  });
})();
